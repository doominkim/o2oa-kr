스크립트 옵션 사용 안내

선택 가능 값 배열을 반환합니다. 예:

return [{
    text: "상품",     //옵션 텍스트
    value: "goods",  //옵션 값
    children: [{     //하위 옵션, children이 없으면 리프 노드
        text: "침구",
        value: "bedding",
        //하위 옵션에도 동적 로딩 함수를 쓸 수 있으며, 문자열은 그 함수 이름입니다.
        children: "getSubOptions"
    }]
}]

선택 가능 값 배열로 resolve되는 Promise 객체를 반환합니다. 예:

return new Promise((resolve)=>{
    window.setTimeout(()=>{
        let items = [{
            text: "상품",
            value: "goods",
            children: [{
                text: "침구",
                value: "bedding"
            }]
        }]
        resolve(items) //옵션 데이터 반환
    }, 500)
})

동적 로딩 옵션 안내

Promise를 반환하는 이름 있는 함수를 반환해야 합니다. 예:

value는 상위 옵션 값 경로입니다. 예: ['상품','침구','베개']. 루트 노드의 경로는 빈 배열 []입니다.

e는 이벤트 객체이며, e.target으로 이벤트가 발생한 Dom 노드를 가져올 수 있습니다.

return function getSubOptions(value, e){
    return new Promise((resolve)=>{
        window.setTimeout(()=>{
            let items;
            const v = value.at(-1); //마지막 항목 반환
            if( !v ){ //비어 있으면 루트 노드
                items = [{
                    text: "상품", //옵션 텍스트
                    value: "goods",//옵션 값
                    //하위 옵션, 반환하는 함수와 같은 이름. 이 노드를 펼치면 이 함수를 다시 호출함을 뜻합니다.
                    children: getSubOptions
                }]
            }else if(v === 'goods'){
                items = [{
                    text: "침구",
                    value: "bedding",
                    children: [{
                        text: "베개",
                        value: "pillow"
                    }]
                }]
            }
            resolve(items) //옵션 데이터 반환
        }, 500)
    })
}

검색 안내

검색어를 매개변수로 받아 옵션 배열 또는 Promise를 반환하는 함수를 반환해야 합니다. 예:

return function(key){
    const options = [];
    if(key.includes('문법')){
        options.push({
            text: "문법과 데이터 형식",
            value: "syntax"
        });
    }
    return options
}
return function(key){
    return new Promise((resolve, reject)=>{
        setTimeout(()=>{
            const options = [];
            if(key.includes('문법')){
                options.push({
                    text: "문법과 데이터 형식",
                    value: "syntax"
                });
            }
            resolve(options);
        }, 1000)
    })
}
